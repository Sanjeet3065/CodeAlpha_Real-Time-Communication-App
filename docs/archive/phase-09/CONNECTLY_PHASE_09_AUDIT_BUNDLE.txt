function extractBearerOrQueryToken(rawUrl: string, authHeader?: string): string | null {
  if (authHeader && authHeader.startsWith('Bearer ')) {
    const token = authHeader.slice('Bearer '.length).trim();
    if (token) return token;
  }
  try {
    const parsed = new URL(rawUrl, 'http://localhost');
    const qToken = parsed.searchParams.get('access_token');
    if (qToken && qToken.trim()) {
      return qToken.trim();
    }
  } catch {
    return null;
  }
  return null;
}

function parseTokenMetadata(
  metadata?: string
): { epoch: number; meetingId: string } | null {
  if (!metadata) return null;
  try {
    const parsed = JSON.parse(metadata) as { epoch?: unknown; meetingId?: unknown };
    if (
      typeof parsed.epoch === 'number' &&
      Number.isInteger(parsed.epoch) &&
      parsed.epoch >= 1 &&
      typeof parsed.meetingId === 'string' &&
      /^[a-f0-9]{24}$/i.test(parsed.meetingId)
    ) {
      return {
        epoch: parsed.epoch,
        meetingId: parsed.meetingId.toLowerCase(),
      };
    }
    return null;
  } catch {
    return null;
  }
}

/**
 * Authoritative MongoDB revalidation of meeting & participant state.
 * Strictly fails CLOSED (`503 LIVEKIT_AUTH_UNAVAILABLE`) on any database error.
 */
export async function revalidateParticipantAccess(
  meetingId: string,
  userId: string,
  tokenEpoch: number,
  tokenNbf?: number
): Promise<RevalidationResult> {
  let meeting: { status: string; expiresAt: Date } | null;
  let participant: {
    admissionStatus: string;
    tokenEpoch?: number;
    tokenRevokedBefore?: Date | null;
  } | null;

  try {
    [meeting, participant] = await Promise.all([
      Meeting.findById(meetingId).select('status expiresAt'),
      MeetingParticipant.findOne({ meetingId, userId }).select(
        'admissionStatus tokenEpoch tokenRevokedBefore'
      ),
    ]);
  } catch (err) {
    logger.error('[LiveKitGateway] MongoDB authorization check failed (fail-closed)', {
      meetingId,
      userId,
      errorCategory: 'LIVEKIT_AUTH_UNAVAILABLE',
      message: err instanceof Error ? err.message : 'Database lookup failed',
    });
    return {
      allowed: false,
      statusCode: 503,
      code: 'LIVEKIT_AUTH_UNAVAILABLE',
      message: 'Authorization service is temporarily unavailable.',
    };
  }

  if (!meeting) {
    return {
      allowed: false,
      statusCode: 404,
      code: 'MEETING_NOT_FOUND',
      message: 'Meeting no longer exists.',
    };
  }

  const now = new Date();
  if (meeting.status === 'ENDED' || meeting.status === 'EXPIRED' || now >= meeting.expiresAt) {
    return {
      allowed: false,
      statusCode: 410,
      code: 'LIVEKIT_MEETING_ENDED',
      message: 'This meeting has ended and no longer accepts media signaling connections.',
    };
  }

  if (!participant) {
    return {
      allowed: false,
      statusCode: 403,
      code: 'LIVEKIT_ACCESS_DENIED',
      message: 'Participant is not authorized for this meeting.',
    };
  }

  if (participant.admissionStatus === 'REMOVED') {
    return {
      allowed: false,
      statusCode: 403,
      code: 'PARTICIPANT_REMOVED',
      message: 'Participant has been removed from this meeting; token access revoked.',
    };
  }

  if (participant.admissionStatus !== 'ADMITTED') {
    return {
      allowed: false,
      statusCode: 403,
      code: 'LIVEKIT_ACCESS_DENIED',
      message: 'Participant is not currently admitted to this meeting.',
    };
  }

  const expectedEpoch =
    typeof participant.tokenEpoch === 'number' && participant.tokenEpoch >= 1
      ? participant.tokenEpoch
      : 1;
  if (tokenEpoch !== expectedEpoch) {
    return {
      allowed: false,
      statusCode: 403,
      code: 'LIVEKIT_TOKEN_REVOKED',
      message: 'LiveKit access token has been revoked.',
    };
  }

  if (participant.tokenRevokedBefore && typeof tokenNbf === 'number') {
    const cutoffSec = Math.floor(participant.tokenRevokedBefore.getTime() / 1000);
    if (tokenNbf < cutoffSec) {
      return {
        allowed: false,
        statusCode: 403,
        code: 'LIVEKIT_TOKEN_REVOKED',
        message: 'LiveKit access token predates revocation cutoff.',
      };
    }
  }

  return { allowed: true };
}

export type AtomicFenceResult =
  | { allowed: true; fenceVersion: number }
  | { allowed: false; statusCode: number; code: string; message: string };

/**
 * MongoDB-backed atomic admission linearization point.
 *
 * Serializes signaling admission against concurrent `removeParticipant` / `leaveMeeting`
 * on the exact same `MeetingParticipant` document via single-document conditional
 * `findOneAndUpdate`:
 * - If removal commits first (`admissionStatus = REMOVED`, `tokenEpoch` incremented),
 *   this conditional update atomically returns `null` (loses the race) and the gateway
 *   NEVER emits HTTP 101.
 * - If admission commits first (`$inc: { signalFenceVersion: 1 }`), admission linearized
 *   before removal; any subsequent removal publishes a durable `LiveKitRevocationEvent`
 *   that terminates the already-registered socket pair via `LiveKitRevocationCoordinator`.
 * - Fails strictly CLOSED (`503 LIVEKIT_FENCE_UNAVAILABLE`) on any MongoDB error.
 */
export async function claimSignalingAdmissionFence(
  meetingId: string,
  userId: string,
  tokenEpoch: number,
  tokenNbf?: number
): Promise<AtomicFenceResult> {
  try {
    const now = new Date();
    const meeting = await Meeting.findById(meetingId).select('status expiresAt').lean();
    if (!meeting) {
      return {
        allowed: false,
        statusCode: 404,
        code: 'MEETING_NOT_FOUND',
        message: 'Meeting no longer exists.',
      };
    }
    if (meeting.status === 'ENDED' || meeting.status === 'EXPIRED' || now >= meeting.expiresAt) {
      return {
        allowed: false,
        statusCode: 410,
        code: 'LIVEKIT_MEETING_ENDED',
        message: 'This meeting has ended and no longer accepts media signaling connections.',
      };
    }

    const filter: Record<string, unknown> = {
      meetingId,
      userId,
      admissionStatus: 'ADMITTED',
      tokenEpoch,
    };
    if (typeof tokenNbf === 'number') {
      // Allow tokens whose nbf (seconds) >= floor(tokenRevokedBefore / 1000)
      filter.$or = [
        { tokenRevokedBefore: null },
        { tokenRevokedBefore: { $exists: false } },
        { tokenRevokedBefore: { $lt: new Date((tokenNbf + 1) * 1000) } },
      ];
    }

    const claimed = await MeetingParticipant.findOneAndUpdate(
      filter,
      { $inc: { signalFenceVersion: 1 } },
      { new: true, projection: { signalFenceVersion: 1 } }
    ).lean();

    if (!claimed) {
      // Admission lost the atomic document race; determine exact authoritative denial reason
      const diag = await revalidateParticipantAccess(meetingId, userId, tokenEpoch, tokenNbf);
      if (!diag.allowed) {
        return diag;
      }
      return {
        allowed: false,
        statusCode: 403,
        code: 'LIVEKIT_TOKEN_REVOKED',
        message: 'Signaling admission lost atomic authorization fence.',
      };
    }

    return {
      allowed: true,
      fenceVersion: typeof claimed.signalFenceVersion === 'number' ? claimed.signalFenceVersion : 1,
    };
  } catch (err) {
    logger.error('[LiveKitGateway] Atomic admission fence failed (fail-closed)', {
      meetingId,
      userId,
      errorCategory: 'LIVEKIT_FENCE_UNAVAILABLE',
      message: err instanceof Error ? err.message : 'Atomic fence database error',
    });
    return {
      allowed: false,
      statusCode: 503,
      code: 'LIVEKIT_FENCE_UNAVAILABLE',
      message: 'Signaling admission fencing service is temporarily unavailable.',
    };
  }
}

/**
 * Authoritative pre-admission verification executed BEFORE any HTTP validate
 * or WebSocket signaling upgrade is forwarded to the self-hosted LiveKit SFU.
 */
export async function authorizeLiveKitSignalHandshake(
  rawUrl: string,
  authHeader?: string,
  livekitSvc: LiveKitService = livekitService,
  revocationCoordinator?: LiveKitRevocationCoordinator
): Promise<SignalAuthResult> {
  if (revocationCoordinator && !revocationCoordinator.isHealthy()) {
    return {
      allowed: false,
      statusCode: 503,
      code: 'LIVEKIT_REVOCATION_UNAVAILABLE',
      message: 'Signaling revocation service is temporarily unavailable.',
    };
  }

  const token = extractBearerOrQueryToken(rawUrl, authHeader);
  if (!token) {
    return {
      allowed: false,
      statusCode: 401,
      code: 'LIVEKIT_MISSING_TOKEN',
      message: 'Missing LiveKit access token.',
    };
  }

  let cfg: LiveKitConfig;
  try {
    cfg = livekitSvc.assertConfigured();
  } catch {
    return {
      allowed: false,
      statusCode: 503,
      code: 'LIVEKIT_NOT_CONFIGURED',
      message: 'LiveKit signaling infrastructure is not configured.',
    };
  }

  let claims: Awaited<ReturnType<TokenVerifier['verify']>>;
  try {
    const verifier = new TokenVerifier(cfg.apiKey, cfg.apiSecret);
    claims = await verifier.verify(token);
  } catch {
    return {
      allowed: false,
      statusCode: 401,
      code: 'LIVEKIT_INVALID_TOKEN',
      message: 'Invalid or expired LiveKit access token.',
    };
  }

  const roomName = claims.video?.room || '';
  const identity = claims.sub || '';
  if (!claims.video?.roomJoin || !roomName || !identity) {
    return {
      allowed: false,
      statusCode: 401,
      code: 'LIVEKIT_INVALID_TOKEN',
      message: 'LiveKit access token lacks valid roomJoin claims.',
    };
  }

  const meetingId = parseLiveKitRoomMeetingId(roomName);
  const userId = parseLiveKitParticipantUserId(identity);
  if (!meetingId || !userId) {
    return {
      allowed: false,
      statusCode: 403,
      code: 'LIVEKIT_ACCESS_DENIED',
      message: 'Token room or identity does not match CONNECTLY authorization policy.',
    };
  }

  const meta = parseTokenMetadata(claims.metadata);
  if (!meta) {
    return {
      allowed: false,
      statusCode: 403,
      code: 'LIVEKIT_TOKEN_REVOKED',
      message: 'LiveKit access token lacks valid signed epoch metadata.',
    };
  }

  if (meta.meetingId !== meetingId) {
    return {
      allowed: false,
      statusCode: 403,
      code: 'LIVEKIT_ACCESS_DENIED',
      message: 'Token meeting metadata does not match room grant.',
    };
  }

  const recheck = await revalidateParticipantAccess(
    meetingId,
    userId,
    meta.epoch,
    typeof claims.nbf === 'number' ? claims.nbf : undefined
  );
  if (!recheck.allowed) {
    return recheck;
  }

  return {
    allowed: true,
    meetingId,
    userId,
    roomName,
    identity,
    tokenEpoch: meta.epoch,
    tokenNbf: typeof claims.nbf === 'number' ? claims.nbf : undefined,
    upstreamConfig: cfg,
  };
}

function writeSocketHttpError(socket: Duplex, statusCode: number, message: string): void {
  if (socket.destroyed) return;
  const statusText = http.STATUS_CODES[statusCode] || 'Error';
  const body = message;
  const response =
    `HTTP/1.1 ${statusCode} ${statusText}\r\n` +
    `Content-Type: text/plain; charset=utf-8\r\n` +
    `Content-Length: ${Buffer.byteLength(body)}\r\n` +
    `Connection: close\r\n` +
    `\r\n` +
    body;
  try {
    socket.end(response);
  } catch {
    socket.destroy();
  }
}

function buildUpstreamHeaders(
  req: IncomingMessage,
  upstreamHost: string
): Record<string, string | string[]> {
  const headers: Record<string, string | string[]> = {
    host: upstreamHost,
    connection: 'Upgrade',
    upgrade: 'websocket',
  };
  const forwardNames = [
    'sec-websocket-key',
    'sec-websocket-version',
    'sec-websocket-extensions',
    'sec-websocket-protocol',
    'origin',
    'user-agent',
    'authorization',
    'x-forwarded-for',
  ];
  for (const name of forwardNames) {
    const val = req.headers[name];
    if (val !== undefined) {
      headers[name] = val;
    }
  }
  return headers;
}

export class LiveKitSignalGateway {
  private readonly livekit: LiveKitService;
  private readonly hooks: LiveKitSignalGatewayHooks;
  private readonly coordinator: LiveKitRevocationCoordinator;
  private readonly entriesByParticipant = new Map<string, Set<TrackedSocketEntry>>();
  private readonly entriesByMeeting = new Map<string, Set<TrackedSocketEntry>>();
  private readonly pendingByParticipant = new Map<string, Set<PendingUpgradeHandle>>();
  private readonly pendingByMeeting = new Map<string, Set<PendingUpgradeHandle>>();
  private readonly unsubEvent: () => void;
  private readonly unsubHealth: () => void;
  private isClosing = false;

  constructor(
    livekitSvc: LiveKitService = livekitService,
    hooks: LiveKitSignalGatewayHooks = {},
    coordinator: LiveKitRevocationCoordinator = livekitRevocationCoordinator
  ) {
    this.livekit = livekitSvc;
    this.hooks = hooks;
    this.coordinator = coordinator;

    this.unsubEvent = this.coordinator.subscribe((event) => {
      this.applyRevocationEvent(event);
    });
    this.unsubHealth = this.coordinator.onHealthChange((healthy, reason) => {
      if (!healthy) {
        this.failClosedOnCoordinatorOutage(reason);
      }
    });
  }

  private participantKey(meetingId: string, userId: string): string {
    return `${meetingId}:${userId}`;
  }

  public getActiveSocketCount(): number {
    let total = 0;
    for (const set of this.entriesByMeeting.values()) {
      total += set.size * 2; // client + upstream pair per tracked connection
    }
    return total;
  }

  public getPendingUpgradeCount(): number {
    let total = 0;
    for (const set of this.pendingByMeeting.values()) {
      total += set.size;
    }
    return total;
  }

  /**
   * Applies a distributed cross-instance revocation event received via MongoDB
   * Change Stream or durable cursor sweep. Terminates ONLY affected pending
   * handshakes and active sockets whose tokenEpoch predates `revokedEpoch`
   * (or all sockets for MEETING_ENDED), leaving Host and unrelated peers untouched.
   */
  public applyRevocationEvent(event: DistributedRevocationPayload): number {
    if (event.kind === 'MEETING_ENDED' || !event.userId) {
      return this.disconnectMeetingSockets(event.meetingId);
    }

    const cutoffEpoch =
      typeof event.revokedEpoch === 'number' && event.revokedEpoch >= 1
        ? event.revokedEpoch
        : null;
    return this.disconnectParticipantSockets(event.meetingId, event.userId, cutoffEpoch);
  }

  private failClosedOnCoordinatorOutage(reason?: string): number {
    let closed = 0;
    const msg = reason || 'Signaling revocation service unavailable; failing closed.';

    for (const set of Array.from(this.pendingByMeeting.values())) {
      for (const handle of Array.from(set)) {
        handle.abort(503, msg);
        closed += 1;
      }
    }
    this.pendingByParticipant.clear();
    this.pendingByMeeting.clear();

    for (const set of Array.from(this.entriesByMeeting.values())) {
      for (const entry of Array.from(set)) {
        closed += this.destroySocketEntry(entry);
      }
    }
    this.entriesByParticipant.clear();
    this.entriesByMeeting.clear();

    return closed;
  }

  private destroySocketEntry(entry: TrackedSocketEntry): number {
    let closed = 0;
    try {
      if (!entry.clientSocket.destroyed) {
        entry.clientSocket.destroy();
        closed += 1;
      }
    } catch {
      // ignore
    }
    try {
      if (!entry.proxySocket.destroyed) {
        entry.proxySocket.destroy();
        closed += 1;
      }
    } catch {
      // ignore
    }
    return closed;
  }

  private trackPendingUpgrade(
    meetingId: string,
    userId: string,
    handle: PendingUpgradeHandle
  ): () => void {
    const pKey = this.participantKey(meetingId, userId);
    let pSet = this.pendingByParticipant.get(pKey);
    if (!pSet) {
      pSet = new Set();
      this.pendingByParticipant.set(pKey, pSet);
    }
    let mSet = this.pendingByMeeting.get(meetingId);
    if (!mSet) {
      mSet = new Set();
      this.pendingByMeeting.set(meetingId, mSet);
    }

    pSet.add(handle);
    mSet.add(handle);

    let cleaned = false;
    return () => {
      if (cleaned) return;
      cleaned = true;
      pSet?.delete(handle);
      if (pSet && pSet.size === 0) this.pendingByParticipant.delete(pKey);
      mSet?.delete(handle);
      if (mSet && mSet.size === 0) this.pendingByMeeting.delete(meetingId);
    };
  }

  private trackSocketEntry(entry: TrackedSocketEntry): () => void {
    const pKey = this.participantKey(entry.meetingId, entry.userId);
    let pSet = this.entriesByParticipant.get(pKey);
    if (!pSet) {
      pSet = new Set();
      this.entriesByParticipant.set(pKey, pSet);
    }
    let mSet = this.entriesByMeeting.get(entry.meetingId);
    if (!mSet) {
      mSet = new Set();
      this.entriesByMeeting.set(entry.meetingId, mSet);
    }

    pSet.add(entry);
    mSet.add(entry);

    let cleaned = false;
    return () => {
      if (cleaned) return;
      cleaned = true;
      pSet?.delete(entry);
      if (pSet && pSet.size === 0) this.entriesByParticipant.delete(pKey);
      mSet?.delete(entry);
      if (mSet && mSet.size === 0) this.entriesByMeeting.delete(entry.meetingId);
    };
  }

  /**
   * Immediately aborts matching pending handshakes AND terminates active proxied
   * signaling sockets for ONLY the specified participant, without affecting
   * Host or any other admitted participant in the room.
   *
   * If `revokedEpoch` is provided, only connections whose `tokenEpoch < revokedEpoch`
   * are terminated (preventing delayed events from disrupting newer re-admitted sessions).
   */
  public disconnectParticipantSockets(
    meetingId: string,
    userId: string,
    revokedEpoch: number | null = null
  ): number {
    const pKey = this.participantKey(meetingId, userId);
    let closed = 0;

    const pending = this.pendingByParticipant.get(pKey);
    const meetingPending = this.pendingByMeeting.get(meetingId);
    if (pending && pending.size > 0) {
      for (const handle of Array.from(pending)) {
        if (revokedEpoch !== null && handle.tokenEpoch >= revokedEpoch) {
          continue;
        }
        handle.abort(403, 'Participant has been removed from this meeting; token access revoked.');
        pending.delete(handle);
        meetingPending?.delete(handle);
        closed += 1;
      }
      if (pending.size === 0) this.pendingByParticipant.delete(pKey);
      if (meetingPending && meetingPending.size === 0) {
        this.pendingByMeeting.delete(meetingId);
      }
    }

    const entries = this.entriesByParticipant.get(pKey);
    const meetingEntries = this.entriesByMeeting.get(meetingId);
    if (entries && entries.size > 0) {
      for (const entry of Array.from(entries)) {
        if (revokedEpoch !== null && entry.tokenEpoch >= revokedEpoch) {
          continue;
        }
        entries.delete(entry);
        meetingEntries?.delete(entry);
        closed += this.destroySocketEntry(entry);
      }
      if (entries.size === 0) this.entriesByParticipant.delete(pKey);
      if (meetingEntries && meetingEntries.size === 0) {
        this.entriesByMeeting.delete(meetingId);
      }
    }

    return closed;
  }

  /**
   * Aborts all pending handshakes and terminates all active proxied signaling
   * sockets belonging to an ended meeting.
   */
  public disconnectMeetingSockets(meetingId: string): number {
    let closed = 0;

    const pending = this.pendingByMeeting.get(meetingId);
    if (pending && pending.size > 0) {
      for (const handle of Array.from(pending)) {
        handle.abort(410, 'This meeting has ended and no longer accepts media signaling connections.');
        closed += 1;
      }
      this.pendingByMeeting.delete(meetingId);
    }
    for (const [pKey, pSet] of Array.from(this.pendingByParticipant.entries())) {
      if (pKey.startsWith(`${meetingId}:`)) {
        pSet.clear();
        this.pendingByParticipant.delete(pKey);
      }
    }

    const entries = this.entriesByMeeting.get(meetingId);
    if (entries && entries.size > 0) {
      for (const entry of Array.from(entries)) {
        closed += this.destroySocketEntry(entry);
      }
      this.entriesByMeeting.delete(meetingId);
    }
    for (const [pKey, pSet] of Array.from(this.entriesByParticipant.entries())) {
      if (pKey.startsWith(`${meetingId}:`)) {
        pSet.clear();
        this.entriesByParticipant.delete(pKey);
      }
    }

    return closed;
  }

  /**
   * Gracefully shuts down the Signaling Admission Gateway: unsubscribes from
   * revocation events, rejects new upgrades, aborts all pending handshakes,
   * and closes all active proxied sockets.
   */
  public close(): number {
    this.isClosing = true;
    this.unsubEvent();
    this.unsubHealth();

    let closed = 0;
    for (const set of Array.from(this.pendingByMeeting.values())) {
      for (const handle of Array.from(set)) {
        handle.abort(503, 'Signaling gateway shutting down.');
        closed += 1;
      }
    }
    this.pendingByParticipant.clear();
    this.pendingByMeeting.clear();

    for (const set of Array.from(this.entriesByMeeting.values())) {
      for (const entry of Array.from(set)) {
        closed += this.destroySocketEntry(entry);
      }
    }
    this.entriesByParticipant.clear();
    this.entriesByMeeting.clear();

    return closed;
  }

  /**
   * Express router handling `HEAD /livekit-signal`, `OPTIONS`, and
   * authoritative pre-admission validation for `/rtc/validate` and `/rtc/v1/validate`.
   */
  public createRouter(): Router {
    const router = Router();

    router.use((_req: Request, res: Response, next) => {
      res.setHeader('Access-Control-Allow-Origin', '*');
      res.setHeader('Access-Control-Allow-Methods', 'GET, HEAD, OPTIONS');
      res.setHeader('Access-Control-Allow-Headers', 'Authorization, Content-Type');
      next();
    });

    router.options('*', (_req: Request, res: Response) => {
      res.status(204).end();
    });

    router.head('/', (_req: Request, res: Response) => {
      if (this.isClosing || !this.coordinator.isHealthy()) {
        res.status(503).end();
        return;
      }
      res.status(200).end();
    });

    router.get('/', (_req: Request, res: Response) => {
      if (this.isClosing || !this.coordinator.isHealthy()) {
        res.status(503).type('text/plain').send('Signaling gateway unavailable.');
        return;
      }
      res.status(200).type('text/plain').send('OK');
    });

    const handleValidate = async (req: Request, res: Response): Promise<void> => {
      if (this.isClosing) {
        res.status(503).type('text/plain').send('Signaling gateway shutting down.');
        return;
      }

      const auth = await authorizeLiveKitSignalHandshake(
        req.originalUrl,
        req.headers.authorization,
        this.livekit,
        this.coordinator
      );

      if (!auth.allowed) {
        logger.warn('[LiveKitGateway] Rejected signaling validate request', {
          statusCode: auth.statusCode,
          code: auth.code,
        });
        res.status(auth.statusCode).type('text/plain').send(auth.message);
        return;
      }

      const upstreamHttpBase = toLiveKitHttpUrl(auth.upstreamConfig.url);
      const strippedUrl = req.originalUrl.slice(LIVEKIT_SIGNAL_GATEWAY_PATH.length) || '/rtc/validate';
      const targetUrl = `${upstreamHttpBase}${strippedUrl}`;

      try {
        const upstreamRes = await fetch(targetUrl, {
          method: 'GET',
          headers: req.headers.authorization
            ? { Authorization: req.headers.authorization }
            : undefined,
        });
        const text = await upstreamRes.text();
        res.status(upstreamRes.status).type('text/plain').send(text);
      } catch {
        res.status(503).type('text/plain').send('LiveKit signaling server is unavailable.');
      }
    };

    router.get('/rtc/validate', (req, res, next) => {
      handleValidate(req, res).catch(next);
    });
    router.get('/rtc/v1/validate', (req, res, next) => {
      handleValidate(req, res).catch(next);
    });

    // Explicitly block any alternate or malformed sub-routes under /livekit-signal/*
    router.all('*', (_req: Request, res: Response) => {
      res.status(404).type('text/plain').send('Unknown LiveKit signaling route.');
    });

    return router;
  }

  /**
   * Attaches an HTTP Upgrade interceptor for `/livekit-signal/rtc` and `/livekit-signal/rtc/v1`.
   * Leaves Socket.IO (`/socket.io`) untouched and rejects any unauthorized or alternate
   * `/rtc*` or `/livekit-signal/*` upgrade paths cleanly.
   */
  public attach(httpServer: HttpServer): void {
    httpServer.on('upgrade', (req: IncomingMessage, clientSocket: Duplex, head: Buffer) => {
      const rawUrl = req.url || '';
      const normalizedPath = normalizeUpgradePath(rawUrl);
      if (!normalizedPath) {
        writeSocketHttpError(clientSocket, 400, 'Malformed signaling request URL.');
        return;
      }

      if (normalizedPath.startsWith('/socket.io')) {
        return;
      }

      clientSocket.on('error', () => {
        // Prevent uncaught socket reset exceptions
      });

      if (
        normalizedPath !== `${LIVEKIT_SIGNAL_GATEWAY_PATH}/rtc` &&
        normalizedPath !== `${LIVEKIT_SIGNAL_GATEWAY_PATH}/rtc/v1`
      ) {
        writeSocketHttpError(clientSocket, 404, 'Unknown WebSocket signaling route.');
        return;
      }

      if (this.isClosing) {
        writeSocketHttpError(clientSocket, 503, 'Signaling gateway shutting down.');
        return;
      }

      if (!this.coordinator.isHealthy()) {
        writeSocketHttpError(clientSocket, 503, 'Signaling revocation service is temporarily unavailable.');
        return;
      }

      this.handleWebSocketUpgrade(req, clientSocket, head).catch((err) => {
        logger.error('[LiveKitGateway] Unexpected signaling upgrade error', {
          message: err instanceof Error ? err.message : 'Unknown upgrade error',
        });
        writeSocketHttpError(clientSocket, 500, 'Signaling gateway error.');
      });
    });
  }

  private async handleWebSocketUpgrade(
    req: IncomingMessage,
    clientSocket: Duplex,
    head: Buffer
  ): Promise<void> {
    const rawUrl = req.url || '';

    if (this.hooks.beforeAuthorize) {
      await this.hooks.beforeAuthorize();
    }
    if (this.isClosing || !this.coordinator.isHealthy() || clientSocket.destroyed) {
      writeSocketHttpError(clientSocket, 503, 'Signaling gateway unavailable.');
      return;
    }

    const auth = await authorizeLiveKitSignalHandshake(
      rawUrl,
      req.headers.authorization,
      this.livekit,
      this.coordinator
    );

    if (!auth.allowed) {
      logger.warn('[LiveKitGateway] Blocked unauthorized/revoked WebSocket upgrade', {
        statusCode: auth.statusCode,
        code: auth.code,
      });
      writeSocketHttpError(clientSocket, auth.statusCode, auth.message);
      return;
    }

    const pendingHandle: PendingUpgradeHandle = {
      aborted: false,
      abortStatusCode: 403,
      abortMessage: 'Participant access revoked.',
      tokenEpoch: auth.tokenEpoch,
      abort: (statusCode = 403, message = 'Participant access revoked.') => {
        if (pendingHandle.aborted) return;
        pendingHandle.aborted = true;
        pendingHandle.abortStatusCode = statusCode;
        pendingHandle.abortMessage = message;
        try {
          pendingHandle.proxyReq?.destroy();
        } catch {
          // ignore
        }
        writeSocketHttpError(clientSocket, statusCode, message);
      },
    };
    const untrackPending = this.trackPendingUpgrade(auth.meetingId, auth.userId, pendingHandle);

    const onClientAbortEarly = () => {
      pendingHandle.aborted = true;
      untrackPending();
      try {
        pendingHandle.proxyReq?.destroy();
      } catch {
        // ignore
      }
    };
    clientSocket.once('close', onClientAbortEarly);

    if (this.hooks.afterAuthorize) {
      await this.hooks.afterAuthorize();
    }
    if (pendingHandle.aborted || this.isClosing || !this.coordinator.isHealthy() || clientSocket.destroyed) {
      untrackPending();
      writeSocketHttpError(
        clientSocket,
        !this.coordinator.isHealthy() ? 503 : pendingHandle.abortStatusCode,
        !this.coordinator.isHealthy() ? 'Signaling revocation service unavailable.' : pendingHandle.abortMessage
      );
      return;
    }

    if (this.hooks.beforeUpstreamConnect) {
      await this.hooks.beforeUpstreamConnect();
    }
    if (pendingHandle.aborted || this.isClosing || !this.coordinator.isHealthy() || clientSocket.destroyed) {
      untrackPending();
      writeSocketHttpError(
        clientSocket,
        !this.coordinator.isHealthy() ? 503 : pendingHandle.abortStatusCode,
        !this.coordinator.isHealthy() ? 'Signaling revocation service unavailable.' : pendingHandle.abortMessage
      );
      return;
    }

    const preUpstreamCheck = await revalidateParticipantAccess(
      auth.meetingId,
      auth.userId,
      auth.tokenEpoch,
      auth.tokenNbf
    );
    if (
      !preUpstreamCheck.allowed ||
      pendingHandle.aborted ||
      this.isClosing ||
      !this.coordinator.isHealthy() ||
      clientSocket.destroyed
    ) {
      untrackPending();
      const status = !this.coordinator.isHealthy()
        ? 503
        : preUpstreamCheck.allowed
          ? pendingHandle.abortStatusCode
          : preUpstreamCheck.statusCode;
      const msg = !this.coordinator.isHealthy()
        ? 'Signaling revocation service unavailable.'
        : preUpstreamCheck.allowed
          ? pendingHandle.abortMessage
          : preUpstreamCheck.message;
      writeSocketHttpError(clientSocket, status, msg);
      return;
    }

    const upstreamBase = new URL(toLiveKitHttpUrl(auth.upstreamConfig.url));
    const parsedClientUrl = new URL(rawUrl, 'http://localhost');
    const normalizedPath = parsedClientUrl.pathname.replace(/\/+$/, '');
    const strippedPathname = normalizedPath.slice(LIVEKIT_SIGNAL_GATEWAY_PATH.length) || '/rtc';
    const strippedPathWithQuery = `${strippedPathname}${parsedClientUrl.search}`;
    const isTls = upstreamBase.protocol === 'https:';
    const transport = isTls ? https : http;

    const proxyReq = transport.request({
      protocol: upstreamBase.protocol,
      hostname: upstreamBase.hostname,
      port: upstreamBase.port || (isTls ? 443 : 80),
      method: 'GET',
      path: strippedPathWithQuery,
      headers: buildUpstreamHeaders(req, upstreamBase.host),
    });
    pendingHandle.proxyReq = proxyReq;

    const duringHandshakePromise = this.hooks.duringUpstreamHandshake
      ? this.hooks.duringUpstreamHandshake()
      : Promise.resolve();

    proxyReq.on('upgrade', (proxyRes, proxySocket, proxyHead) => {
      proxySocket.pause();
      proxySocket.on('error', () => {
        // prevent uncaught upstream socket errors
      });

      const finalizeUpgrade = async () => {
        await duringHandshakePromise;

        if (this.hooks.afterUpstreamHandshake) {
          await this.hooks.afterUpstreamHandshake();
        }

        if (
          pendingHandle.aborted ||
          this.isClosing ||
          !this.coordinator.isHealthy() ||
          clientSocket.destroyed ||
          proxySocket.destroyed
        ) {
          untrackPending();
          if (!proxySocket.destroyed) proxySocket.destroy();
          writeSocketHttpError(
            clientSocket,
            !this.coordinator.isHealthy() ? 503 : pendingHandle.abortStatusCode,
            !this.coordinator.isHealthy() ? 'Signaling revocation service unavailable.' : pendingHandle.abortMessage
          );
          void this.livekit.removeParticipant(auth.meetingId, auth.userId);
          return;
        }

        if (this.hooks.duringSocketRegister) {
          await this.hooks.duringSocketRegister();
        }

        if (
          pendingHandle.aborted ||
          this.isClosing ||
          !this.coordinator.isHealthy() ||
          clientSocket.destroyed ||
          proxySocket.destroyed
        ) {
          untrackPending();
          if (!proxySocket.destroyed) proxySocket.destroy();
          writeSocketHttpError(
            clientSocket,
            !this.coordinator.isHealthy() ? 503 : pendingHandle.abortStatusCode,
            !this.coordinator.isHealthy() ? 'Signaling revocation service unavailable.' : pendingHandle.abortMessage
          );
          void this.livekit.removeParticipant(auth.meetingId, auth.userId);
          return;
        }

        // Register socket entry BEFORE post-handshake revalidation and BEFORE 101
        // so both local disconnectParticipantSockets() AND cross-instance coordinator
        // events can terminate the socket pair at any point before, during, or after 101.
        clientSocket.off('close', onClientAbortEarly);
        const socketEntry: TrackedSocketEntry = {
          meetingId: auth.meetingId,
          userId: auth.userId,
          tokenEpoch: auth.tokenEpoch,
          clientSocket,
          proxySocket,
        };
        const untrackSockets = this.trackSocketEntry(socketEntry);

        const cleanupBoth = () => {
          untrackPending();
          untrackSockets();
          if (!clientSocket.destroyed) clientSocket.destroy();
          if (!proxySocket.destroyed) proxySocket.destroy();
        };

        clientSocket.once('error', cleanupBoth);
        proxySocket.once('error', cleanupBoth);
        clientSocket.once('end', cleanupBoth);
        proxySocket.once('end', cleanupBoth);
        clientSocket.once('close', cleanupBoth);
        proxySocket.once('close', cleanupBoth);

        // Post-handshake authoritative MongoDB revalidation
        const postHandshakeCheck = await revalidateParticipantAccess(
          auth.meetingId,
          auth.userId,
          auth.tokenEpoch,
          auth.tokenNbf
        );

        if (
          !postHandshakeCheck.allowed ||
          pendingHandle.aborted ||
          this.isClosing ||
          !this.coordinator.isHealthy() ||
          clientSocket.destroyed ||
          proxySocket.destroyed
        ) {
          untrackPending();
          untrackSockets();
          if (!proxySocket.destroyed) proxySocket.destroy();
          const status = !this.coordinator.isHealthy()
            ? 503
            : postHandshakeCheck.allowed
              ? pendingHandle.abortStatusCode
              : postHandshakeCheck.statusCode;
          const msg = !this.coordinator.isHealthy()
            ? 'Signaling revocation service unavailable.'
            : postHandshakeCheck.allowed
              ? pendingHandle.abortMessage
              : postHandshakeCheck.message;
          logger.warn('[LiveKitGateway] Post-handshake revalidation rejected upgraded socket', {
            meetingId: auth.meetingId,
            userId: auth.userId,
            statusCode: status,
            code: postHandshakeCheck.allowed ? 'GATEWAY_ABORTED' : postHandshakeCheck.code,
          });
          writeSocketHttpError(clientSocket, status, msg);
          void this.livekit.removeParticipant(auth.meetingId, auth.userId);
          return;
        }

        // Deterministic hook boundary: AFTER postHandshakeCheck succeeded, BEFORE syncNow() & atomic fence
        if (this.hooks.beforeWrite101) {
          await this.hooks.beforeWrite101();
        }

        // Pre-101 Distributed Sync & Atomic Linearization Fence:
        // 1. Fail closed if coordinator became unhealthy/stalled.
        // 2. Synchronize distributed revocation cursor (`syncNow`).
        // 3. Complete final MongoDB authorization read (`finalReadCheck`).
        // 4. Execute atomic MongoDB single-document linearization point (`claimSignalingAdmissionFence`)
        //    strictly serialized against `removeParticipant` / `leaveMeeting` on `MeetingParticipant`.
        if (!this.coordinator.isHealthy() || this.isClosing) {
          untrackPending();
          untrackSockets();
          if (!proxySocket.destroyed) proxySocket.destroy();
          writeSocketHttpError(clientSocket, 503, 'Signaling revocation service unavailable.');
          void this.livekit.removeParticipant(auth.meetingId, auth.userId);
          return;
        }

        if (this.coordinator.isRunning()) {
          const synced = await this.coordinator.syncNow();
          if (!synced || !this.coordinator.isHealthy()) {
            untrackPending();
            untrackSockets();
            if (!proxySocket.destroyed) proxySocket.destroy();
            writeSocketHttpError(clientSocket, 503, 'Signaling revocation sync failed (fail-closed).');
            void this.livekit.removeParticipant(auth.meetingId, auth.userId);
            return;
          }
        }

        const finalReadCheck = await revalidateParticipantAccess(
          auth.meetingId,
          auth.userId,
          auth.tokenEpoch,
          auth.tokenNbf
        );

        if (
          !finalReadCheck.allowed ||
          pendingHandle.aborted ||
          this.isClosing ||
          !this.coordinator.isHealthy() ||
          clientSocket.destroyed ||
          proxySocket.destroyed
        ) {
          untrackPending();
          untrackSockets();
          if (!proxySocket.destroyed) proxySocket.destroy();
          const status = !this.coordinator.isHealthy()
            ? 503
            : finalReadCheck.allowed
              ? pendingHandle.abortStatusCode
              : finalReadCheck.statusCode;
          const msg = !this.coordinator.isHealthy()
            ? 'Signaling revocation service unavailable.'
            : finalReadCheck.allowed
              ? pendingHandle.abortMessage
              : finalReadCheck.message;
          logger.warn('[LiveKitGateway] Pre-101 read check rejected upgraded socket', {
            meetingId: auth.meetingId,
            userId: auth.userId,
            statusCode: status,
            code: finalReadCheck.allowed ? 'GATEWAY_ABORTED' : finalReadCheck.code,
          });
          writeSocketHttpError(clientSocket, status, msg);
          void this.livekit.removeParticipant(auth.meetingId, auth.userId);
          return;
        }

        // Deterministic barrier AFTER syncNow() AND AFTER final MongoDB read, BEFORE atomic fence
        if (this.hooks.afterSyncBeforeAtomicFence) {
          await this.hooks.afterSyncBeforeAtomicFence();
        }

        // ATOMIC LINEARIZATION POINT: conditional single-document findOneAndUpdate on MeetingParticipant
        const fenceClaim = await claimSignalingAdmissionFence(
          auth.meetingId,
          auth.userId,
          auth.tokenEpoch,
          auth.tokenNbf
        );

        if (
          !fenceClaim.allowed ||
          pendingHandle.aborted ||
          this.isClosing ||
          !this.coordinator.isHealthy() ||
          clientSocket.destroyed ||
          proxySocket.destroyed
        ) {
          untrackPending();
          untrackSockets();
          if (!proxySocket.destroyed) proxySocket.destroy();
          const status = !this.coordinator.isHealthy()
            ? 503
            : fenceClaim.allowed
              ? pendingHandle.abortStatusCode
              : fenceClaim.statusCode;
          const msg = !this.coordinator.isHealthy()
            ? 'Signaling revocation service unavailable.'
            : fenceClaim.allowed
              ? pendingHandle.abortMessage
              : fenceClaim.message;
          logger.warn('[LiveKitGateway] Atomic admission fence rejected upgraded socket', {
            meetingId: auth.meetingId,
            userId: auth.userId,
            statusCode: status,
            code: fenceClaim.allowed ? 'GATEWAY_ABORTED' : fenceClaim.code,
          });
          writeSocketHttpError(clientSocket, status, msg);
          void this.livekit.removeParticipant(auth.meetingId, auth.userId);
          return;
        }

        // Deterministic barrier AFTER atomic linearization point, immediately BEFORE clientSocket.write(HTTP 101)
        if (this.hooks.afterAtomicFenceBeforeSocketWrite101) {
          await this.hooks.afterAtomicFenceBeforeSocketWrite101();
        }

        untrackPending();

        if (
          pendingHandle.aborted ||
          this.isClosing ||
          !this.coordinator.isHealthy() ||
          clientSocket.destroyed ||
          proxySocket.destroyed
        ) {
          untrackSockets();
          if (!proxySocket.destroyed) proxySocket.destroy();
          writeSocketHttpError(
            clientSocket,
            !this.coordinator.isHealthy() ? 503 : pendingHandle.abortStatusCode,
            !this.coordinator.isHealthy()
              ? 'Signaling revocation service unavailable.'
              : pendingHandle.abortMessage
          );
          void this.livekit.removeParticipant(auth.meetingId, auth.userId);
          return;
        }

        const statusLine = `HTTP/1.1 ${proxyRes.statusCode || 101} ${proxyRes.statusMessage || 'Switching Protocols'}\r\n`;
        const headerLines: string[] = [];
        for (const [key, val] of Object.entries(proxyRes.headers)) {
          if (val === undefined) continue;
          if (Array.isArray(val)) {
            for (const v of val) {
              headerLines.push(`${key}: ${v}\r\n`);
            }
          } else {
            headerLines.push(`${key}: ${val}\r\n`);
          }
        }

        clientSocket.write(`${statusLine}${headerLines.join('')}\r\n`);
        if (proxyHead && proxyHead.length > 0) {
          clientSocket.write(proxyHead);
        }
        if (head && head.length > 0) {
          proxySocket.write(head);
        }

        proxySocket.resume();
        clientSocket.pipe(proxySocket);
        proxySocket.pipe(clientSocket);

        if (this.hooks.afterWrite101) {
          await this.hooks.afterWrite101();
        }
        if (this.hooks.afterSocketRegister) {
          await this.hooks.afterSocketRegister();
        }
      };

      finalizeUpgrade().catch((err) => {
        untrackPending();
        if (!proxySocket.destroyed) proxySocket.destroy();
        logger.error('[LiveKitGateway] Error during post-handshake upgrade finalization', {
          meetingId: auth.meetingId,
          userId: auth.userId,
          message: err instanceof Error ? err.message : 'Unknown upgrade finalization error',
        });
        writeSocketHttpError(clientSocket, 500, 'Signaling handshake finalization error.');
      });
    });

    proxyReq.on('response', (proxyRes) => {
      untrackPending();
      const chunks: Buffer[] = [];
      proxyRes.on('data', (chunk: Buffer) => chunks.push(chunk));
      proxyRes.on('end', () => {
        const body =
          Buffer.concat(chunks).toString('utf8') || 'Upstream signaling rejected connection.';
        writeSocketHttpError(clientSocket, proxyRes.statusCode || 502, body);
      });
    });

    proxyReq.on('error', () => {
      untrackPending();
      if (!pendingHandle.aborted) {
        writeSocketHttpError(clientSocket, 503, 'LiveKit signaling server is unavailable.');
      }
    });

    proxyReq.end();
  }
}

/**
 * Distributed cross-instance revocation coordinator backed by MongoDB.
 *
 * Guarantees & Properties:
 * - Global Monotonic Cursor (`seq`): Immune to cross-host ObjectId clock skew.
 * - Zero Skipped Events: `lastContiguousSeq` advances only across contiguous `seq`
 *   values (holding a bounded grace window for in-flight out-of-order commits while
 *   still dispatching newly observed higher-`seq` events immediately).
 * - Deduplication & Out-of-Order Safety: Duplicate events are dropped via `dispatchedSeqs`;
 *   out-of-order events are delivered safely without advancing past missing lower `seq`s.
 * - Change Stream Resume & Fallback: Captures `resumeAfter` tokens on Replica Sets,
 *   reconnects with `resumeAfter`, and executes immediate durable cursor catch-up on error.
 * - TTL Retention Guard: 24-hour event TTL (24x max 1-hour JWT TTL) plus explicit
 *   retention-gap detection (`oldestAvailableSeq > lastContiguousSeq + 1` -> fail-closed).
 * - Stalled Delivery Detection: `isHealthy()` checks both DB failure count AND
 *   `Date.now() - lastSuccessfulSyncAt > maxStaleDeliveryMs` so stalled delivery
 *   triggers fail-closed protection even when no DB exception is thrown.
 */
export class LiveKitRevocationCoordinator {
  private readonly pollIntervalMs: number;
  private readonly enableChangeStream: boolean;
  private readonly maxConsecutiveFailures: number;
  private readonly maxStaleDeliveryMs: number;
  private readonly gapSkipGraceMs: number;
  private readonly initialCursorSeq?: number;

  private readonly eventListeners = new Set<RevocationListener>();
  private readonly healthListeners = new Set<RevocationHealthListener>();

  private running = false;
  private healthy = true;
  private deliveryPaused = false;
  private unavailableReason: string | null = null;
  private transportMode: 'stopped' | 'change_stream' | 'cursor_poll' = 'stopped';
  private consecutiveFailures = 0;
  private lastSuccessfulSyncAt = Date.now();

  private lastContiguousSeq = 0;
  private readonly dispatchedSeqs = new Set<number>();
  private readonly gapFirstSeenAt = new Map<number, number>();
  private lastResumeToken: unknown = null;

  private pollTimer: ReturnType<typeof setInterval> | null = null;
  private reconnectTimer: ReturnType<typeof setTimeout> | null = null;
  private changeStream: { close: () => Promise<void> } | null = null;
  private inflightSync: Promise<boolean> | null = null;

  constructor(options: LiveKitRevocationCoordinatorOptions = {}) {
    this.pollIntervalMs = options.pollIntervalMs ?? 250;
    this.enableChangeStream = options.enableChangeStream ?? true;
    this.maxConsecutiveFailures = options.maxConsecutiveFailures ?? 2;
    this.maxStaleDeliveryMs =
      options.maxStaleDeliveryMs ?? Math.max(5_000, this.pollIntervalMs * 10);
    this.gapSkipGraceMs = options.gapSkipGraceMs ?? 5_000;
    this.initialCursorSeq = options.initialCursorSeq;
    if (typeof options.initialCursorSeq === 'number' && options.initialCursorSeq >= 0) {
      this.lastContiguousSeq = options.initialCursorSeq;
    }
  }

  public isRunning(): boolean {
    return this.running;
  }

  public getLastContiguousSeq(): number {
    return this.lastContiguousSeq;
  }

  public getTransportMode(): 'stopped' | 'change_stream' | 'cursor_poll' {
    return this.transportMode;
  }

  public getUnavailableReason(): string | null {
    return this.unavailableReason;
  }

  public getLastResumeToken(): unknown {
    return this.lastResumeToken;
  }

  /**
   * Checks BOTH explicit health state AND delivery-staleness watchdog.
   * If running and delivery has stalled beyond `maxStaleDeliveryMs`, transitions
   * immediately to UNHEALTHY (fail-closed) and notifies subscribed gateways.
   */
  public isHealthy(): boolean {
    if (!this.healthy) return false;
    if (this.running && Date.now() - this.lastSuccessfulSyncAt > this.maxStaleDeliveryMs) {
      this.setHealth(false, 'Revocation delivery stalled beyond staleness threshold.');
      return false;
    }
    return true;
  }

  public subscribe(listener: RevocationListener): () => void {
    this.eventListeners.add(listener);
    return () => {
      this.eventListeners.delete(listener);
    };
  }

  public onHealthChange(listener: RevocationHealthListener): () => void {
    this.healthListeners.add(listener);
    return () => {
      this.healthListeners.delete(listener);
    };
  }

  /**
   * Pauses background notification delivery (used by deterministic security tests
   * to simulate delayed cross-instance revocation delivery without arbitrary sleeps).
   */
  public pauseDelivery(): void {
    this.deliveryPaused = true;
  }

  public resumeDelivery(): void {
    this.deliveryPaused = false;
  }

  private setHealth(nextHealthy: boolean, reason?: string): void {
    const prev = this.healthy;
    this.healthy = nextHealthy;
    this.unavailableReason = nextHealthy ? null : reason || 'Revocation coordinator unavailable.';
    if (prev !== nextHealthy) {
      if (!nextHealthy) {
        logger.error('[LiveKitRevocation] Coordinator transitioned to UNHEALTHY (fail-closed)', {
          reason: this.unavailableReason,
        });
      } else {
        logger.info('[LiveKitRevocation] Coordinator recovered to HEALTHY');
      }
      for (const listener of Array.from(this.healthListeners)) {
        try {
          listener(nextHealthy, this.unavailableReason ?? undefined);
        } catch {
          // isolate subscriber errors
        }
      }
    }
  }

  public markUnavailable(reason = 'Revocation notification channel disconnected.'): void {
    this.setHealth(false, reason);
  }

  /**
   * Manually forces a staleness timestamp for deterministic watchdog tests.
   */
  public setLastSuccessfulSyncAtForTest(timestampMs: number): void {
    this.lastSuccessfulSyncAt = timestampMs;
  }

  /**
   * Delivers a single revocation event with exact deduplication and monotonic
   * contiguous-sequence tracking (safe against duplicate and out-of-order arrival).
   */
  public ingestEvent(payload: DistributedRevocationPayload): boolean {
    if (this.dispatchedSeqs.has(payload.seq)) {
      return false;
    }
    this.dispatchedSeqs.add(payload.seq);
    this.gapFirstSeenAt.delete(payload.seq);

    // Advance contiguous cursor strictly across consecutive observed sequences
    while (this.dispatchedSeqs.has(this.lastContiguousSeq + 1)) {
      this.lastContiguousSeq += 1;
    }

    // Bound memory of dispatchedSeqs by pruning entries strictly behind lastContiguousSeq
    if (this.dispatchedSeqs.size > 4000) {
      for (const s of Array.from(this.dispatchedSeqs)) {
        if (s <= this.lastContiguousSeq) {
          this.dispatchedSeqs.delete(s);
        }
      }
    }

    for (const listener of Array.from(this.eventListeners)) {
      try {
        listener(payload);
      } catch (err) {
        logger.error('[LiveKitRevocation] Listener error while applying revocation', {
          eventId: payload.eventId,
          seq: payload.seq,
          message: err instanceof Error ? err.message : 'Unknown listener error',
        });
      }
    }
    return true;
  }

  private advanceExpiredGaps(maxObservedSeq: number): void {
    const now = Date.now();
    for (let missing = this.lastContiguousSeq + 1; missing < maxObservedSeq; missing += 1) {
      if (this.dispatchedSeqs.has(missing)) continue;
      const firstSeen = this.gapFirstSeenAt.get(missing);
      if (firstSeen === undefined) {
        this.gapFirstSeenAt.set(missing, now);
      } else if (now - firstSeen >= this.gapSkipGraceMs) {
        // Gap exceeded in-flight commit grace period (aborted counter allocation); advance past it
        this.gapFirstSeenAt.delete(missing);
        this.dispatchedSeqs.add(missing);
      }
    }
    while (this.dispatchedSeqs.has(this.lastContiguousSeq + 1)) {
      this.lastContiguousSeq += 1;
    }
  }

  public async start(): Promise<void> {
    if (this.running) return;
    this.running = true;
    this.consecutiveFailures = 0;

    try {
      if (typeof this.initialCursorSeq === 'number' && this.initialCursorSeq >= 0) {
        this.lastContiguousSeq = this.initialCursorSeq;
      } else {
        const latest = await LiveKitRevocationEvent.findOne({})
          .sort({ seq: -1 })
          .select('seq')
          .lean();
        this.lastContiguousSeq = latest && typeof latest.seq === 'number' ? latest.seq : 0;
      }
      this.lastSuccessfulSyncAt = Date.now();
      this.setHealth(true);
    } catch (err) {
      this.setHealth(
        false,
        err instanceof Error ? err.message : 'Unable to initialize revocation cursor.'
      );
    }

    if (this.enableChangeStream) {
      await this.tryAttachChangeStream();
    } else {
      this.transportMode = 'cursor_poll';
    }

    this.pollTimer = setInterval(() => {
      if (this.deliveryPaused) {
        // Still run staleness watchdog while delivery is paused
        this.isHealthy();
        return;
      }
      void this.syncNow();
    }, this.pollIntervalMs);
    if (typeof this.pollTimer.unref === 'function') {
      this.pollTimer.unref();
    }
  }

  private async tryAttachChangeStream(): Promise<void> {
    try {
      const db = LiveKitRevocationEvent.db;
      const admin = db?.db?.admin();
      if (admin) {
        const hello = await admin.command({ hello: 1 });
        if (!hello?.setName) {
          this.transportMode = 'cursor_poll';
          return;
        }
      }

      const watchOpts: Record<string, unknown> = { fullDocument: 'updateLookup' };
      if (this.lastResumeToken) {
        watchOpts.resumeAfter = this.lastResumeToken;
      }

      const stream = LiveKitRevocationEvent.watch(
        [{ $match: { operationType: 'insert' } }],
        watchOpts
      );

      stream.on('change', (change: any) => {
        if (change?._id) {
          this.lastResumeToken = change._id;
        }
        if (this.deliveryPaused) return;
        const doc = change?.fullDocument;
        if (!doc || !doc._id || !doc.meetingId || !doc.kind || typeof doc.seq !== 'number') return;
        this.consecutiveFailures = 0;
        this.lastSuccessfulSyncAt = Date.now();
        this.setHealth(true);
        this.ingestEvent({
          eventId: String(doc._id),
          seq: doc.seq,
          meetingId: String(doc.meetingId),
          userId: doc.userId ? String(doc.userId) : null,
          kind: doc.kind,
          revokedEpoch: typeof doc.revokedEpoch === 'number' ? doc.revokedEpoch : null,
          fenceVersion: typeof doc.fenceVersion === 'number' ? doc.fenceVersion : null,
          reason: typeof doc.reason === 'string' ? doc.reason : null,
          createdAt: doc.createdAt ? new Date(doc.createdAt) : new Date(),
        });
      });

      stream.on('error', (err: any) => {
        logger.warn(
          '[LiveKitRevocation] Change stream interrupted; running durable cursor catch-up and scheduling resume',
          {
            message: err instanceof Error ? err.message : String(err || 'Change stream error'),
          }
        );
        this.transportMode = 'cursor_poll';
        this.changeStream = null;
        if (!this.deliveryPaused) {
          void this.syncNow();
        }
        if (this.running && !this.reconnectTimer) {
          this.reconnectTimer = setTimeout(() => {
            this.reconnectTimer = null;
            if (this.running) {
              void this.tryAttachChangeStream();
            }
          }, 1_000);
          if (typeof this.reconnectTimer.unref === 'function') {
            this.reconnectTimer.unref();
          }
        }
      });

      this.changeStream = stream;
      this.transportMode = 'change_stream';
    } catch {
      this.transportMode = 'cursor_poll';
      this.changeStream = null;
    }
  }

  /**
   * Synchronously sweeps all durable `LiveKitRevocationEvent` records with
   * `seq > lastContiguousSeq`, checks TTL retention continuity (fail-closed if
   * undelivered events expired during an extended outage), and dispatches events
   * in exact sequence order.
   */
  public async syncNow(): Promise<boolean> {
    if (this.inflightSync) {
      return this.inflightSync;
    }

    const run = async (): Promise<boolean> => {
      try {
        const events = await LiveKitRevocationEvent.find({
          seq: { $gt: this.lastContiguousSeq },
        })
          .sort({ seq: 1 })
          .limit(500)
          .lean();

        this.consecutiveFailures = 0;
        this.lastSuccessfulSyncAt = Date.now();

        if (events.length > 0) {
          // Retention discontinuity guard: if the oldest event in DB is strictly
          // older than expected AND created before our retention window, detect
          // potential TTL expiry of an undelivered event and fail closed.
          const firstSeq = events[0].seq;
          if (
            this.lastContiguousSeq > 0 &&
            firstSeq > this.lastContiguousSeq + 1 &&
            Date.now() - new Date(events[0].createdAt).getTime() > this.gapSkipGraceMs * 4
          ) {
            this.setHealth(
              false,
              `Revocation cursor discontinuity detected (expected seq ${this.lastContiguousSeq + 1}, oldest available ${firstSeq}); failing closed.`
            );
            return false;
          }

          let maxObservedSeq = this.lastContiguousSeq;
          for (const doc of events) {
            if (typeof doc.seq !== 'number') continue;
            if (doc.seq > maxObservedSeq) maxObservedSeq = doc.seq;
            this.ingestEvent({
              eventId: String(doc._id),
              seq: doc.seq,
              meetingId: String(doc.meetingId),
              userId: doc.userId ? String(doc.userId) : null,
              kind: doc.kind,
              revokedEpoch: typeof doc.revokedEpoch === 'number' ? doc.revokedEpoch : null,
              fenceVersion: typeof doc.fenceVersion === 'number' ? doc.fenceVersion : null,
              reason: typeof doc.reason === 'string' ? doc.reason : null,
              createdAt: doc.createdAt ? new Date(doc.createdAt) : new Date(),
            });
          }
          this.advanceExpiredGaps(maxObservedSeq);
        }

        if (this.running) {
          this.setHealth(true);
        }
        return true;
      } catch (err) {
        this.consecutiveFailures += 1;
        if (this.consecutiveFailures >= this.maxConsecutiveFailures) {
          this.setHealth(
            false,
            err instanceof Error ? err.message : 'Revocation cursor sync failed.'
          );
        }
        return false;
      }
    };

    this.inflightSync = run().finally(() => {
      this.inflightSync = null;
    });
    return this.inflightSync;
  }

  public async stop(): Promise<void> {
    this.running = false;
    this.deliveryPaused = false;
    this.transportMode = 'stopped';
    if (this.pollTimer) {
      clearInterval(this.pollTimer);
      this.pollTimer = null;
    }
    if (this.reconnectTimer) {
      clearTimeout(this.reconnectTimer);
      this.reconnectTimer = null;
    }
    if (this.changeStream) {
      const stream = this.changeStream;
      this.changeStream = null;
      await stream.close().catch(() => {});
    }
    this.eventListeners.clear();
    this.healthListeners.clear();
  }
}

  if (rawSignalMode !== 'auto' && rawSignalMode !== 'true' && rawSignalMode !== 'false') {
    return {
      valid: false,
      reason: 'LIVEKIT_SIGNAL_GATEWAY_ENABLED must be one of auto, true, or false.',
    };
  }
  const signalGatewayEnabled: 'auto' | 'true' | 'false' = rawSignalMode;
  const apiKey = (config.apiKey || '').trim();
  const apiSecret = (config.apiSecret || '').trim();
  const tokenTtlSeconds =
    typeof config.tokenTtlSeconds === 'number' && Number.isFinite(config.tokenTtlSeconds)
      ? Math.floor(config.tokenTtlSeconds)
      : 600;

  if (!url || !apiKey || !apiSecret) {
    return {
      valid: false,
      reason: 'Missing LIVEKIT_URL, LIVEKIT_API_KEY, or LIVEKIT_API_SECRET.',
    };
  }

  let parsed: URL;
  try {
    parsed = new URL(url);
  } catch {
    return { valid: false, reason: 'LIVEKIT_URL is not a valid URL.' };
  }

  const hostLower = parsed.hostname.toLowerCase();
  const isCloud = hostLower.endsWith('.livekit.cloud');
  const isLoopback = hostLower === '127.0.0.1' || hostLower === 'localhost' || hostLower === '::1';

  if (publicUrl) {
    let parsedPub: URL;
    try {
      parsedPub = new URL(publicUrl);
    } catch {
      return { valid: false, reason: 'LIVEKIT_PUBLIC_URL is not a valid URL.' };
    }
    const allowedPubProtocols =
      nodeEnv === 'production' ? ['wss:', 'https:'] : ['ws:', 'wss:', 'http:', 'https:'];
    if (!allowedPubProtocols.includes(parsedPub.protocol)) {
      return {
        valid: false,
        reason:
          nodeEnv === 'production'
            ? 'Production LIVEKIT_PUBLIC_URL must use a secure wss:// or https:// scheme.'
            : 'LIVEKIT_PUBLIC_URL must use ws://, wss://, http://, or https://.',
      };
    }
  }

  if (nodeEnv === 'production' && !isCloud && signalGatewayEnabled === 'false') {
    return {
      valid: false,
      reason:
        'Production self-hosted LiveKit cannot disable LIVEKIT_SIGNAL_GATEWAY_ENABLED (would bypass authoritative JWT revocation).',
    };
  }

  const allowedProtocols =
    nodeEnv === 'production'
      ? publicUrl && isLoopback
        ? ['ws:', 'wss:', 'http:', 'https:']
        : ['wss:', 'https:']
      : ['ws:', 'wss:', 'http:', 'https:'];
  if (!allowedProtocols.includes(parsed.protocol)) {
    return {
      valid: false,
      reason:
        nodeEnv === 'production'
          ? 'Production LIVEKIT_URL must use a secure wss:// or https:// scheme (or private loopback paired with a wss:// LIVEKIT_PUBLIC_URL).'
          : 'LIVEKIT_URL must use ws://, wss://, http://, or https://.',
    };
  }

  const keyLower = apiKey.toLowerCase();
  const secretLower = apiSecret.toLowerCase();
  if (
    PLACEHOLDER_PREFIXES.some(
      (prefix) => keyLower.startsWith(prefix) || secretLower.startsWith(prefix)
    )
  ) {
    return {
      valid: false,
      reason: 'LiveKit credentials contain unconfigured placeholder values.',
    };
  }

  if (nodeEnv === 'production' && (keyLower === 'devkey' || secretLower === 'secret')) {
    return {
      valid: false,
      reason: 'Production deployment cannot use default LiveKit dev credentials.',
    };
  }

  if (tokenTtlSeconds < 60 || tokenTtlSeconds > 3600) {
    return {
      valid: false,
      reason: 'LIVEKIT_TOKEN_TTL_SECONDS must be between 60 and 3600 seconds.',
    };
  }

  return {
    valid: true,
    resolved: {
      url: toLiveKitWsUrl(url),
      publicUrl: publicUrl ? toLiveKitWsUrl(publicUrl) : undefined,
      signalGatewayEnabled,
      apiKey,
      apiSecret,
      tokenTtlSeconds,
    },
  };
}

function isNotFoundLiveKitError(err: unknown): boolean {
  if (!err || typeof err !== 'object') return false;
  const e = err as { status?: number; code?: string | number; message?: string };
  if (e.status === 404 || e.code === 'not_found' || e.code === 404) return true;
  const msg = String(e.message || '').toLowerCase();
  return msg.includes('not found') || msg.includes('does not exist') || msg.includes('room not found');
}

function isAlreadyExistsLiveKitError(err: unknown): boolean {
  if (!err || typeof err !== 'object') return false;
  const e = err as { status?: number; code?: string | number; message?: string };
  if (e.status === 409 || e.code === 'already_exists') return true;
  const msg = String(e.message || '').toLowerCase();
  return msg.includes('already exists');
}

  /**
   * Disconnects a removed or departed participant from the LiveKit SFU room.
   * Best-effort with sanitized error logging so CONNECTLY DB state remains authoritative.
   */
  public async removeParticipant(
    meetingId: string,
    userId: string,
    requestId?: string
  ): Promise<boolean> {
    if (!this.isConfigured()) {
      return false;
    }

    const cfg = this.assertConfigured();
    const roomName = getLiveKitRoomName(meetingId);
    const participantIdentity = getLiveKitParticipantIdentity(userId);
    const client = this.getRoomClient(cfg);

    // Pass revokeTokenTs (now + 1s) so LiveKit Cloud natively invalidates any
    // token whose nbf <= now (including same-second tokens). Self-hosted OSS
    // LiveKit Server v1.13.11 ignores revokeTokenTs and only disconnects active
    // sessions; pre-admission revocation on self-hosted is enforced by
    // livekitSignalGateway against authoritative MongoDB state + tokenEpoch.
    const revokeTokenTs = BigInt(Math.floor(Date.now() / 1000) + 1);

    try {
      await client.removeParticipant(roomName, participantIdentity, { revokeTokenTs });
      logger.info('[LiveKit] Evicted participant from SFU room', {
        requestId,
        meetingId,
        roomName,
        participantIdentity,
        revokeTokenTs: revokeTokenTs.toString(),
      });
      return true;
    } catch (err) {
      if (isNotFoundLiveKitError(err)) {
        return true;
      }
      logger.warn('[LiveKit] Failed to evict participant from SFU room (non-fatal)', {
        requestId,
        meetingId,
        roomName,
        participantIdentity,
        errorCategory: 'LIVEKIT_UNAVAILABLE',
        message: err instanceof Error ? err.message : 'Unknown eviction error',
      });
      return false;
    }
  }

  /**
   * Closes/deletes the LiveKit SFU room when a CONNECTLY meeting ends.
   * Performs 1 bounded retry on transient failure; falls back safely to LiveKit's
   * emptyTimeout reconciliation while logging safe diagnostic metadata.
   */
  public async deleteRoom(meetingId: string, requestId?: string): Promise<boolean> {
    if (!this.isConfigured()) {
      return false;
    }

    const cfg = this.assertConfigured();
    const roomName = getLiveKitRoomName(meetingId);
    const client = this.getRoomClient(cfg);

    for (let attempt = 1; attempt <= 2; attempt += 1) {
      try {
        await client.deleteRoom(roomName);
        logger.info('[LiveKit] Deleted SFU room upon meeting termination', {
          requestId,
          meetingId,
          roomName,
          attempt,
        });
        return true;
      } catch (err) {
        if (isNotFoundLiveKitError(err)) {
          return true;
        }
        logger.warn('[LiveKit] Room cleanup attempt failed', {
          requestId,
          meetingId,
          roomName,
          attempt,
          errorCategory: 'LIVEKIT_UNAVAILABLE',
          message: err instanceof Error ? err.message : 'Unknown room deletion error',
        });
      }
    }

    return false;
  }
}

function createBarrier() {
  let markArrived!: () => void;
  let releaseWaiter!: () => void;

    markArrived = r;
  });
  const gate = new Promise<void>((r) => {

interface WsAttemptResult {
  statusCode: number;
  socket?: Duplex;
  closedPromise?: Promise<void>;
}

function attemptWsUpgrade(port: number, reqPath: string): Promise<WsAttemptResult> {
  return new Promise((resolve) => {
    const key = crypto.randomBytes(16).toString('base64');
    const req = http.request({
      hostname: '127.0.0.1',
      port,
      path: reqPath,
      method: 'GET',
      headers: {
        Connection: 'Upgrade',
        Upgrade: 'websocket',
        'Sec-WebSocket-Key': key,
        'Sec-WebSocket-Version': '13',
      },
    });

    let settled = false;
    const finish = (result: WsAttemptResult) => {
      if (settled) return;
      settled = true;
      resolve(result);
    };

    req.on('upgrade', (res, socket) => {
      socket.on('error', () => {});
      const closedPromise = new Promise<void>((r) => {
        if (socket.destroyed) r();
        else socket.once('close', () => r());
      });
      finish({ statusCode: res.statusCode || 101, socket, closedPromise });
    });

    req.on('response', (res) => {
      res.resume();
      finish({ statusCode: res.statusCode || 0 });
    });

    req.on('error', () => {
      finish({ statusCode: 0 });
    });

    req.end();
  });
}

function createBarrier() {
  let markArrived!: () => void;
  let releaseWaiter!: () => void;
  const arrived = new Promise<void>((r) => {
    markArrived = r;
  });
  const gate = new Promise<void>((r) => {
    releaseWaiter = r;
  });
  return {
    arrived,
    release: () => releaseWaiter(),
    pause: async () => {
      markArrived();
      await gate;
    },
  };
}

function createCountdownBarrier(expectedCount: number) {
  let arrivedCount = 0;
  let markAllArrived!: () => void;
  let releaseAll!: () => void;
  const allArrived = new Promise<void>((r) => {
    markAllArrived = r;
  });
  const gate = new Promise<void>((r) => {
    releaseAll = r;
  });
  return {
    allArrived,
    release: () => releaseAll(),
    pause: async () => {
      arrivedCount += 1;
      if (arrivedCount >= expectedCount) {
        markAllArrived();
      }
      await gate;
    },
  };
}

interface WsAttemptResult {
  statusCode: number;
  socket?: Duplex;
  closedPromise?: Promise<void>;
}

function attemptWsUpgrade(port: number, reqPath: string): Promise<WsAttemptResult> {
  return new Promise((resolve) => {
    const key = crypto.randomBytes(16).toString('base64');
    const req = http.request({
      hostname: '127.0.0.1',
      port,
      path: reqPath,
      method: 'GET',
      headers: {
        Connection: 'Upgrade',
        Upgrade: 'websocket',
        'Sec-WebSocket-Key': key,
        'Sec-WebSocket-Version': '13',
      },
    });

    let settled = false;
    const finish = (result: WsAttemptResult) => {
      if (settled) return;
      settled = true;
      resolve(result);
    };

    req.on('upgrade', (res, socket) => {
      socket.on('error', () => {});
      const closedPromise = new Promise<void>((r) => {
        if (socket.destroyed) r();
        else socket.once('close', () => r());
      });
      finish({ statusCode: res.statusCode || 101, socket, closedPromise });
    });

    req.on('response', (res) => {
      res.resume();
      finish({ statusCode: res.statusCode || 0 });
    });

    req.on('error', () => {
      finish({ statusCode: 0 });
    });

    req.end();
  });
}

# CONNECTLY — Phase 09 Security Audit Report

**Phase:** 09 – WebRTC + LiveKit Integration  
**Audit Date:** 2026-11-18 (Updated following independent Removed-Participant Token Reuse & `revokeTokenTs` Security Audit)  
**Deployed & Verified Versions:** `livekit-server-sdk@2.20.0` (backend), `livekit-client@2.24.1` (frontend), self-hosted `livekit-server version 1.13.11` (OSS binary compiled from `github.com/livekit/livekit@v1.13.11`).

## 1. Executive Security Finding: Self-Hosted LiveKit `v1.13.11` Token Revocation Audit

### 1.1 Source-Code & Experimental Investigation (`livekit-server-sdk@2.20.0` vs. `livekit-server v1.13.11`)

An independent security audit inspected both the installed SDK (`livekit-server-sdk@2.20.0`) and the compiled LiveKit Server `v1.13.11` Go source code, and executed live multi-browser token-reuse experiments:

1. **SDK Protocol Support (`livekit-server-sdk@2.20.0`):** `RoomServiceClient.removeParticipant(room, identity, { revokeTokenTs?: bigint })` exposes `revokeTokenTs` (`int64 revoke_token_ts = 3` in `livekit.RoomParticipantIdentity`).
2. **Self-Hosted Server Reality (`github.com/livekit/livekit@v1.13.11`):**
   - Searching the entire `livekit@v1.13.11` Go source tree for `RevokeTokenTs` / `revoke_token_ts` returned **0 occurrences**.
   - `pkg/service/roomservice.go:226-247` checks `HasParticipant` (returning `404 participant does not exist` when offline, even when `revokeTokenTs` is supplied) and forwards to `roommanager.go:861-870`, which only disconnects the active peer (`SERVICE_REQUEST_REMOVE_PARTICIPANT`) without storing any revocation cutoff.
   - `pkg/service/auth.go:86-117` (`APIKeyAuthMiddleware`) verifies only JWT HMAC signature, `nbf`, and `exp` via `v.Verify(secret)` — there is no revocation table in OSS `v1.13.11`.
3. **Experimental Pre-Mitigation Proof (Raw `ws://127.0.0.1:7880`):**
   - Host removed Guest via CONNECTLY's Phase 08 `DELETE /api/v1/meetings/:code/participants/:id` (`200 OK`).
   - Guest was disconnected (`Disconnected`, reason `4` = `PARTICIPANT_REMOVED`) and denied new CONNECTLY tokens (`403 PARTICIPANT_REMOVED`).
   - **HOWEVER**, reconnecting directly to raw `ws://127.0.0.1:7880` using the **exact original unexpired JWT** succeeded (`reconnected: true`, `remoteCount: 1`) in:
     - Same browser context (without requesting a new token).
     - Fresh isolated Chrome browser context.
     - After calling `removeParticipant(room, identity, { revokeTokenTs: now + 5s })` while online (LiveKit logged `"revokeTokenTs": 1794970853`, returned `200`, and still started a new RTC session 2 seconds later with the old JWT).
     - After a full browser disconnect + `removeParticipant(..., { revokeTokenTs })` while offline (LiveKit returned `404 participant does not exist`, and reconnecting with the original JWT succeeded).

### 1.2 Implemented Mitigation: Authoritative Signaling Admission Gateway + Signed `tokenEpoch` + Cloud `revokeTokenTs`

To block old-token reuse **immediately (0 ms post-join race window)** on self-hosted LiveKit **without disrupting Host or any other admitted participant**:

1. **Authoritative LiveKit Signaling Admission Gateway ([livekitSignalGateway.ts](file:///c:/Users/SANJEET%20CHAUHAN/Desktop/code%20Alpha/antigravity%20communication%20application/server/src/realtime/livekitSignalGateway.ts)):**
   - Mounted on CONNECTLY's HTTP/WebSocket server at `/livekit-signal` (`HEAD /livekit-signal`, `GET /livekit-signal/rtc[/v1]/validate`, and WebSocket `UPGRADE /livekit-signal/rtc[/v1]`), leaving `/socket.io` untouched.
   - Automatically enabled (`LIVEKIT_SIGNAL_GATEWAY_ENABLED=auto`) for all self-hosted LiveKit deployments (non-`*.livekit.cloud` hosts); browsers receive `livekitUrl: ws(s)://<api-host>/livekit-signal`.
   - **Pre-Admission Gate (executes BEFORE any packet reaches LiveKit SFU):** Cryptographically verifies the LiveKit JWT (`TokenVerifier`), validates opaque room/identity mapping, queries authoritative MongoDB (`Meeting` + `MeetingParticipant`), and rejects with HTTP `403` / `410` if:
     - Meeting is `ENDED` or `EXPIRED` (`410 LIVEKIT_MEETING_ENDED`).
     - Participant is `REMOVED` (`403 PARTICIPANT_REMOVED`) or not `ADMITTED` (`403 LIVEKIT_ACCESS_DENIED`).
     - Signed JWT `metadata.epoch` does not match MongoDB `MeetingParticipant.tokenEpoch` (`403 LIVEKIT_TOKEN_REVOKED`).
     - JWT `nbf` predates `MeetingParticipant.tokenRevokedBefore` (`403 LIVEKIT_TOKEN_REVOKED`).
   - **Per-Participant Socket Termination:** On removal or leave, only the removed/departed participant's active proxied signaling sockets are closed (`disconnectParticipantSockets`), leaving Host and all other admitted participants connected with **zero disruption**.
   - **Zero JWT Log Leakage:** [requestLogger.ts](file:///c:/Users/SANJEET%20CHAUHAN/Desktop/code%20Alpha/antigravity%20communication%20application/server/src/middlewares/requestLogger.ts) redacts `access_token`, `token`, and `join_request` query parameters before logging.
2. **Monotonic Signed `tokenEpoch` & Post-Generation TOCTOU Guard:**
   - `MeetingParticipant` stores `tokenEpoch` (default `1`, atomically `$inc: 1` on `removeParticipant` and `leaveMeeting`) and `tokenRevokedBefore`.
   - Every LiveKit JWT embeds `epoch` inside HMAC-signed `metadata`, and `issueLiveKitToken` re-verifies `admissionStatus === 'ADMITTED'` AND `tokenEpoch === expectedTokenEpoch` after room provisioning before returning any token.
   - Permanently invalidates pre-departure unexpired JWTs even if a user leaves and legitimately rejoins within the same second.
3. **LiveKit Cloud `revokeTokenTs` Compatibility:**
   - `LiveKitService.removeParticipant` forwards `{ revokeTokenTs: BigInt(Math.floor(Date.now() / 1000) + 1) }` so LiveKit Cloud deployments also benefit from native Cloud control-plane token revocation.

| --- | --- | --- | --- |
| **Participant disconnected** | **YES — Immediate** (`removeParticipant` + per-participant gateway socket teardown; zero disruption to other peers) | **YES — Immediate** (while online; `404` when offline) | **YES — Immediate** |
| **Existing JWT revoked** | **YES — Immediate at Gateway** (`admissionStatus=REMOVED` + `tokenEpoch` mismatch + `tokenRevokedBefore` checked pre-handshake) | **NO — Not supported by OSS `v1.13.11`** (`revokeTokenTs` ignored) | **YES — Cloud control plane** (via `revokeTokenTs`, 10-min Cloud window) |
| **Existing JWT still valid** | **NO at Gateway** (cryptographically signed, but permanently rejected with HTTP `403` pre-upgrade) | **YES** (remains valid at raw `:7880` until `exp`, default `600s`) | **NO** (rejected by Cloud `nbf < revoke_token_ts` cutoff) |
| **New token request denied** | **YES — Immediate** (`403 PARTICIPANT_REMOVED`; stale in-flight requests also rejected) | **YES — Immediate** (`403 PARTICIPANT_REMOVED` from CONNECTLY API) | **YES — Immediate** (`403 PARTICIPANT_REMOVED`) |
| **Old-token reconnection** | **DENIED (HTTP `403`)** across same browser, auto-reconnect, fresh browser context, and offline-then-removed | **STILL POSSIBLE** if client network can directly reach raw `:7880` | **DENIED** by LiveKit Cloud control plane |

> **Mandatory Self-Hosted Production Network Boundary:** Because self-hosted LiveKit Server `v1.13.11` itself lacks native JWT revocation, production self-hosted deployments **MUST** bind LiveKit's raw HTTP/WS signaling port (`7880`) to a private loopback/VPC interface reachable **only** by the CONNECTLY Signaling Admission Gateway (never directly from untrusted client browsers). Only ICE/WebRTC media ports (`7881/TCP`, UDP media range) are exposed, and WebRTC media cannot be established without completing the gateway-authorized `/rtc` signaling handshake.

### 1.4 Final Gateway Race-Condition, Bypass & Fail-Closed Audit (7-Boundary Verification)

A focused follow-up audit identified and resolved four subtle gateway concurrency and deployment edge cases:

1. **7-Boundary Handshake-vs-Removal TOCTOU Race (Fixed & Barrier-Tested):**
   - *Vulnerability found:* Previously, if `removeParticipant` ran after initial gateway authorization but before upstream WebSocket upgrade completed, `disconnectParticipantSockets` saw zero registered sockets and the late-completing upstream `101` could leave a removed participant connected.
   - *Fix implemented ([livekitSignalGateway.ts](file:///c:/Users/SANJEET%20CHAUHAN/Desktop/code%20Alpha/antigravity%20communication%20application/server/src/realtime/livekitSignalGateway.ts)):*
     - Tracks every in-flight handshake (`pendingByParticipant` / `pendingByMeeting`) immediately upon initial authorization so `disconnectParticipantSockets` / `disconnectMeetingSockets` / `close()` aborts pending upstream requests instantly.
     - Executes an authoritative pre-upstream MongoDB re-check AND a **mandatory post-handshake MongoDB revalidation** (`revalidateParticipantAccess`) AFTER registering `(clientSocket, proxySocket)` and BEFORE sending `HTTP/1.1 101 Switching Protocols` to the client.
     - If removal committed on the same instance OR on another API instance (cross-instance MongoDB), the upgraded upstream socket is destroyed, HTTP `403`/`410` is returned (never `101`), and best-effort `livekit.removeParticipant` is invoked.
   - *Verified deterministically ([livekitGatewaySecurity.test.ts](file:///c:/Users/SANJEET%20CHAUHAN/Desktop/code%20Alpha/antigravity%20communication%20application/server/src/tests/livekitGatewaySecurity.test.ts) Tests 1–2):* All 7 exact boundaries (`beforeAuthorize`, `afterAuthorize`, `beforeUpstreamConnect`, `duringUpstreamHandshake`, `afterUpstreamHandshake`, `duringSocketRegister`, `afterSocketRegister`), cross-instance removal, and a simultaneous 5x connection burst with the same old JWT leave **0** active removed sockets while Host remains connected uninterrupted.
2. **Half-Open Socket Cleanup & Graceful Shutdown (Fixed & Verified):**
   - Added `'end'` listeners alongside `'close'`/`'error'` so TCP FIN from either peer immediately destroys and untracks both sockets (`getActiveSocketCount() === 0`).
   - Added `livekitSignalGateway.close()` and wired it into `server/src/server.ts`'s graceful `SIGTERM`/`SIGINT` shutdown handler (`getActiveSocketCount() === 0`, subsequent upgrades receive `503`).
3. **Alternate Route Locking & Fail-Closed MongoDB (Fixed & Verified):**
   - All alternate WebSocket upgrade routes (`/rtc`, `/rtc/v1`, `/livekit-signal/rtc/v2`, `/livekit-signal/*`) and unknown HTTP routes return `404` without reaching LiveKit (leaving `/socket.io` untouched).
   - MongoDB query errors inside `authorizeLiveKitSignalHandshake` / `revalidateParticipantAccess` strictly fail CLOSED (`503 LIVEKIT_AUTH_UNAVAILABLE`).
   - Missing/malformed `epoch` metadata, cross-meeting metadata mismatches, and outsider tokens are rejected (`403`).
4. **Anti-Bypass Production Transport Enforcement (Fixed & Verified):**
   - `env.ts` and `validateLiveKitConfig` reject `LIVEKIT_SIGNAL_GATEWAY_ENABLED=false` for self-hosted production deployments (preventing silent gateway bypass) and enforce `wss://` for `LIVEKIT_PUBLIC_URL` / production gateway URLs.
   - Removed client-side `VITE_LIVEKIT_URL` direct fallback in `useLiveKitConnection.ts` and `livekitClient.ts` so browsers strictly use the authoritative backend `livekitUrl`.
   - Local development verification (real loopback `127.0.0.1:7880` behind `localhost:5000/livekit-signal`) is explicitly distinguished from production infrastructure verification (TLS reverse proxy + VPC port `7880` firewall isolation, documented in `PHASE_09_LIVEKIT_SETUP.md` and verified during Phase 18 Deployment).

### 1.5 Cross-Instance Distributed Revocation, Atomic Linearization & Post-Sync 101 Fencing Audit

A deep distributed-systems audit examined both the post-`syncNow()` / post-final-read `HTTP 101` race and cross-instance event delivery semantics across independent Gateway & API nodes:

1. **Atomic MongoDB Linearization Point (`claimSignalingAdmissionFence` vs. `removeParticipant`):**
   - *Key distributed-systems fact:* Neither a MongoDB read followed by an OS socket `write(101)` nor `coordinator.syncNow()` alone provides cross-instance linearizability.
   - *Implemented linearization mechanism ([livekitSignalGateway.ts](file:///c:/Users/SANJEET%20CHAUHAN/Desktop/code%20Alpha/antigravity%20communication%20application/server/src/realtime/livekitSignalGateway.ts), [MeetingParticipant.model.ts](file:///c:/Users/SANJEET%20CHAUHAN/Desktop/code%20Alpha/antigravity%20communication%20application/server/src/models/MeetingParticipant.model.ts), [meeting.service.ts](file:///c:/Users/SANJEET%20CHAUHAN/Desktop/code%20Alpha/antigravity%20communication%20application/server/src/services/meeting.service.ts)):*
     - Both signaling admission (`claimSignalingAdmissionFence`) and participant removal (`removeParticipant`) execute an **atomic single-document conditional `findOneAndUpdate`** on the exact same `MeetingParticipant` MongoDB document, incrementing monotonic `signalFenceVersion`.
     - **Case A — Removal linearizes BEFORE Admission (including after `syncNow()` and after final MongoDB read, even while revocation delivery is paused):** `claimSignalingAdmissionFence` atomically returns `null` (loses the document race). The gateway destroys the upstream socket, returns `403`, and **NEVER emits `HTTP 101`**.
     - **Case B — Admission linearizes BEFORE Removal:** Admission won the MongoDB document lock with `signalFenceVersion = k`; removal subsequently commits `admissionStatus = 'REMOVED'`, `tokenEpoch = E + 1`, `signalFenceVersion = k + 1`, and publishes a durable `LiveKitRevocationEvent`. Because the socket pair was already registered before the linearization point, `LiveKitRevocationCoordinator` deterministically terminates the established socket upon event delivery. **Unavoidable propagation interval (disclosed honestly):** If a gateway process stalls *after* winning `claimSignalingAdmissionFence` and *before* kernel `socket.write(101)` while asynchronous revocation delivery is delayed, `HTTP 101` reflects the pre-removal linearized admission and is then unconditionally terminated once the durable revocation event propagates to that gateway.
     - **Fail-Closed Fence:** If `claimSignalingAdmissionFence` encounters any MongoDB error, it returns `503 LIVEKIT_FENCE_UNAVAILABLE` and never emits `HTTP 101`.
2. **Durable Global-Sequence Event Delivery (`LiveKitRevocationCounter` + `LiveKitRevocationEvent` + `LiveKitRevocationCoordinator`):**
   - **Clock-Skew-Independent Monotonic `seq`:** Every revocation event atomically allocates a global integer `seq` (`1, 2, 3, ...`) from `LiveKitRevocationCounter`, preventing skipped events caused by cross-host `ObjectId` clock skew.
   - **Zero Skipped Events & Out-of-Order Safety:** `lastContiguousSeq` advances strictly across contiguous sequences (holding a bounded grace window for in-flight out-of-order commits while dispatching higher-`seq` events immediately with exact `dispatchedSeqs` deduplication). Epoch fencing (`entry.tokenEpoch < event.revokedEpoch`) prevents delayed old-epoch events from disrupting newer re-admitted sessions.
   - **Change Stream Resume & Standalone Polling (Separately Defined):**
     - *MongoDB Replica Set (Production):* Real-time push via Change Stream (`watch`), capturing `resumeAfter` tokens for reconnect recovery and running periodic durable `seq` cursor catch-up (`pollIntervalMs`, default `250ms`). Delivery latency is determined by MongoDB oplog commit + network RTT + event-loop scheduling (no unmeasured hard millisecond SLA is claimed).
     - *Standalone MongoDB (Local Dev / CI):* Automatically uses indexed durable `seq` cursor sweeps (`pollIntervalMs`, default `250ms`) with no Change Stream dependency.
   - **TTL Retention & Stalled-Delivery Watchdog (Fail-Closed):**
     - `LiveKitRevocationEvent` TTL is **24 hours (`86400s` — 24x max 1-hour JWT TTL, 144x default 10-min TTL)** so no unexpired JWT can outlive its revocation event; cursor discontinuity detection fails closed if an extended outage causes a retention gap.
     - `coordinator.isHealthy()` enforces both `maxConsecutiveFailures` (`2`) AND a **stalled-delivery watchdog** (`Date.now() - lastSuccessfulSyncAt > maxStaleDeliveryMs`), failing closed (`503` + active socket termination) even if delivery stalls without throwing DB exceptions.
   - *Verified deterministically ([livekitCrossInstanceRevocation.test.ts](file:///c:/Users/SANJEET%20CHAUHAN/Desktop/code%20Alpha/antigravity%20communication%20application/server/src/tests/livekitCrossInstanceRevocation.test.ts) 7/7 PASS & [livekitPostSyncFencing.test.ts](file:///c:/Users/SANJEET%20CHAUHAN/Desktop/code%20Alpha/antigravity%20communication%20application/server/src/tests/livekitPostSyncFencing.test.ts) 6/6 PASS).*
